// Temporary audit harness: renders screens to PNG. Not part of the repo.
import 'dart:io';
import 'dart:ui' as ui;

import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';

const _fontDir = '/data/tools/flutter/bin/cache/artifacts/material_fonts';
final outDir = Platform.environment['SHOT_DIR']!;
final shotKey = GlobalKey();

Future<void> loadFonts() async {
  Future<ByteData> f(String n) async =>
      ByteData.sublistView(File('$_fontDir/$n').readAsBytesSync());
  final roboto = FontLoader('Roboto');
  for (final w in ['Regular', 'Medium', 'Bold', 'Black', 'Light', 'Italic']) {
    roboto.addFont(f('Roboto-$w.ttf'));
  }
  await roboto.load();
  final icons = FontLoader('MaterialIcons')..addFont(f('MaterialIcons-Regular.otf'));
  await icons.load();
}

void phone(WidgetTester tester) {
  tester.view.physicalSize = const Size(1080, 2400);
  tester.view.devicePixelRatio = 3;
  addTearDown(tester.view.reset);
}

Widget framed(Widget app) => RepaintBoundary(key: shotKey, child: app);

Future<void> settle(WidgetTester tester, [int frames = 30]) async {
  for (var i = 0; i < frames; i++) {
    await tester.pump(const Duration(milliseconds: 50));
  }
}

Future<void> shot(WidgetTester tester, String name) async {
  await settle(tester);
  final boundary =
      shotKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
  await tester.runAsync(() async {
    final image = await boundary.toImage(pixelRatio: 3);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File('$outDir/$name.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}

Future<bool> tapText(WidgetTester tester, String text) async {
  final f = find.text(text);
  if (f.evaluate().isEmpty) {
    // ignore: avoid_print
    print('SHOT: missing "$text"');
    return false;
  }
  await tester.ensureVisible(f.first);
  await tester.tap(f.first, warnIfMissed: false);
  await settle(tester, 20);
  return true;
}

void dumpTexts(WidgetTester tester, String label) {
  final texts = find
      .byType(Text)
      .evaluate()
      .map((e) => (e.widget as Text).data)
      .whereType<String>()
      .toList();
  // ignore: avoid_print
  print('SHOT[$label]: ${texts.join(' | ')}');
}
